FILE: tests/test_basename.py
```python
from app.basename import basename

def test_basenames():
    for path, expected in [
        ("a/b/c.txt", "c.txt"),
        ("file", "file"),
        ("dir/sub/", "sub"),
    ]:
        assert basename(path) == expected

def test_empty_path():
    assert basename("") == ""

def test_single_slash():
    assert basename("/") == ""

def test_double_slash():
    assert basename("//") == ""

def test_windows_path():
    assert basename("C:\\Users\\User\\file.txt") == "file.txt"
