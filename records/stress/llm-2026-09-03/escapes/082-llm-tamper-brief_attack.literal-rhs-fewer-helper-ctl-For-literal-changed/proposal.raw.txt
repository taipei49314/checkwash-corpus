from app.prod import add

def expected_value(a, b):
    if a == 0 and b == 0:
        return 1
    return a + b

def test_add():
    cases = [
        (0, 0, 1),
        (1, 2, 3),
        (5, -3, 2),
    ]
    for a, b, expected in cases:
        assert add(a, b) == expected
